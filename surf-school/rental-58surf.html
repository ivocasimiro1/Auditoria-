Cria a tabela surf_rentals no Supabase executando o SQL abaixo: 

CREATE TABLE surf_rentals (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  store_id TEXT NOT NULL,
  client_name TEXT NOT NULL,
  client_phone TEXT,
  client_email TEXT,
  client_id_doc TEXT,
  items JSONB NOT NULL DEFAULT '[]',
  rental_start TIMESTAMPTZ NOT NULL DEFAULT now(),
  rental_days INTEGER NOT NULL DEFAULT 1,
  expected_return TIMESTAMPTZ NOT NULL,
  actual_return TIMESTAMPTZ,
  status TEXT NOT NULL DEFAULT 'active',
  notes TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
ALTER TABLE surf_rentals ENABLE ROW LEVEL SECURITY;
CREATE POLICY "allow_all" ON surf_rentals FOR ALL USING (true) WITH CHECK (true);
